cmake_minimum_required(VERSION 3.20)
project(FHGNative LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "FHG native controller is Windows-only")
endif()
add_executable(FHG WIN32 FHG.cpp resources/FHG.rc)
set_source_files_properties(resources/FHG.rc PROPERTIES LANGUAGE RC)
target_compile_features(FHG PRIVATE cxx_std_17)
target_compile_definitions(FHG PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(FHG PRIVATE Shell32 Comdlg32 Iphlpapi Ws2_32 Ole32)
if(MSVC)
  target_compile_options(FHG PRIVATE /W4 /EHsc)
endif()
set_target_properties(FHG PROPERTIES OUTPUT_NAME FHG)
